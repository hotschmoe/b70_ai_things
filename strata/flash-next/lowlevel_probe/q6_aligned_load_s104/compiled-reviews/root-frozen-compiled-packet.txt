    const uintptr_t a = reinterpret_cast<uintptr_t>(p);

    const bool half = (a & 2) != 0;
    const uint32_t d[5] = {uint32_t(q[0]), uint32_t(q[1]), uint32_t(q[2]), uint32_t(q[3]), uint32_t(q[4])};

}
